nano wordcount.py

from pyspark.sql import SparkSession

spark = SparkSession.builder.appName("WordCount").getOrCreate()

sc = spark.sparkContext

text_data = sc.textFile("input.txt")

word_counts = text_data.flatMap(lambda line: line.split()) \
    .map(lambda word: (word, 1)) \
    .reduceByKey(lambda a, b: a + b)

for word, count in word_counts.collect():
    print(f"{word}: {count}")

spark.stop()

from mpi4py import MPI
comm = MPI.COMM_WORLD
rank = comm.Get_rank()
my_val = (rank + 1) * 10
total_sum = comm.reduce(my_val, op=MPI.SUM, root=0)
if rank == 0:
    print(f"Total Sum: {total_sum}")

size = comm.Get_size()
my_val = rank + 1

4.Parallel sum

nano omp_sum.c


#include <stdio.h>
#include <omp.h>

int main() {
    int sum = 0;
    int n = 10;

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= n; i++) {
        sum += i;
    }

    printf("Total Sum: %d\n", sum);
    return 0;
}

gcc -fopenmp omp_sum.c -o omp_sum
./omp_sum


5.Parallel factorial

nano omp_factorial.c

#include <stdio.h>
#include <omp.h>
int main() {
    int n = 5;
    long long fact = 1;

    #pragma omp parallel for reduction(*:fact)
    for (int i = 1; i <= n; i++) {
        fact *= i;
    }

    printf("Factorial of %d: %lld\n", n, fact);
    return 0;
}


gcc -fopenmp omp_factorial.c -o omp_factorial
./omp_factorial


6.Parallel array addition

nano omp_array.c


#include <stdio.h>
#include <omp.h>

int main() {
    int n = 5;
    int a[] = {1, 2, 3, 4, 5};
    int b[] = {10, 20, 30, 40, 50};
    int result[5];

    #pragma omp parallel for
    for (int i = 0; i < n; i++) {
        result[i] = a[i] + b[i];
    }

    printf("Result Array:\n");
    for (int i = 0; i < n; i++) {
        printf("%d + %d = %d\n", a[i], b[i], result[i]);
    }

    return 0;
}


gcc -fopenmp omp_array.c -o omp_array
OMP_NUM_THREADS=4 ./omp_array
